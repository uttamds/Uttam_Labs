<!DOCTYPE html>
<html>
<head>
    <title>JavaScript Array Demo</title>
</head>
<body>

    <h2>JavaScript Array Demo</h2>
    <button onclick="showArray()">Show Array Details</button>

    <div id="output"></div>

    <script>
        function showArray() {
            let fruits = ["Apple", "Banana", "Mango"];

            let result = "";

            result += "Original Array: " + fruits + "<br><br>";
            result += "First Fruit: " + fruits[0] + "<br>";
            result += "Second Fruit: " + fruits[1] + "<br><br>";

            fruits.push("Orange");

            result += "After Adding Orange: " + fruits + "<br><br>";

            result += "<b>All Fruits:</b><br>";

            for (let i = 0; i < fruits.length; i++) {
                result +=  fruits[i] + "<br>";
            }

            result += "After popping the last item" + "<br>";

            fruits.pop();
          

            for (let i = 0; i < fruits.length; i++) {
                result +=  fruits[i] + "<br>";
            }

            
            document.getElementById("output").innerHTML = result;
        }
    </script>

</body>
</html>

<!DOCTYPE html>
<html>
<head>
    <title>Array Demo</title>
</head>
<body>

    <button onclick="showItem()">Show Item</button>

    <ol id="myList"></ol>

    <ol>
        <li>Car</li>
    </ol>

    <script>
        let fruits = ["Apple", "Mango", "Orange"];
        let index = 0;

        function showItem() {
            if (index < fruits.length) {
                
                let li = document.createElement("li");
                li.innerHTML ="<strong>" +fruits[index] ;
                document.getElementById("myList").appendChild(li);
                index++;
            } else {
                alert("All items are displayed.");
            }
        }
    </script>

</body>
</html>

<!DOCTYPE html>

        }
    </style>
</head>
<body>

    <button onclick="showItem()">Show Item</button>

    <ol id="myList"></ol>

    <div id="message" class="info-band">
        ✓ All items have been displayed.
    </div>

    <script>
        let fruits = ["Apple", "Mango", "Orange"];
        let index = 0;

        function showItem() {
            if (index < fruits.length) {
                let li = document.createElement("li");
                li.innerHTML = fruits[index];
                document.getElementById("myList").appendChild(li);
                index++;
            }

            if (index === fruits.length) {
                document.getElementById("message").style.display = "block";
            }
        }
    </script>

</body>
