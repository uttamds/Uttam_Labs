<!DOCTYPE html>

    <title>JavaScript Array Demo</title>
</head>
<body>

    <h2>JavaScript Array Demo</h2>
    <button onclick="showArray()">Show Array Details</button>

    <div id="output"></div>

    <script>
        function showArray() {
            let fruits = ["Apple", "Banana", "Mango"];

            let result = "";

            result += "Original Array: " + fruits + "<br><br>";
            result += "First Fruit: " + fruits[0] + "<br>";
            result += "Second Fruit: " + fruits[1] + "<br><br>";

            fruits.push("Orange");

            result += "After Adding Orange: " + fruits + "<br><br>";

            result += "<b>All Fruits:</b><br>";

            for (let i = 0; i < fruits.length; i++) {
                result +=  fruits[i] + "<br>";
            }

            result += "After popping the last item" + "<br>";

            fruits.pop();
          

            for (let i = 0; i < fruits.length; i++) {
                result +=  fruits[i] + "<br>";
            }

            
            document.getElementById("output").innerHTML = result;
        }
    </script>

</body>
