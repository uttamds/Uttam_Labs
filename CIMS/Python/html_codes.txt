<!DOCTYPE html>
<html>
<body>

<h2>Fruit List</h2>

<button id="btn" onclick="showFruit()">Show Next Fruit</button>

<p id="result"></p>

<script>
let fruits = ["Apple", "Mango", "Orange"];
let index = 0;

function showFruit() {

    document.getElementById("result").innerHTML += fruits[index] + "<br>";

    // Last fruit?
    if (index == fruits.length - 1) {

        document.getElementById("result").innerHTML +=
            "<br><b>This was the last fruit.</b>";

        // Disable button for 5 seconds
        document.getElementById("btn").disabled = true;

        setTimeout(function () {
            document.getElementById("result").innerHTML = "";
            index = 0;
            document.getElementById("btn").disabled = false;
        }, 5000);

    } else {
        index++;
    }
}
</script>

</body>
</html>
