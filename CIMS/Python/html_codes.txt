<!DOCTYPE html>
<html>
<body>

<h2>Fruit List</h2>

<button onclick="showFruit()">Show Next Fruit</button>

<p id="result"></p>

<script>
let fruits = ["Apple", "Mango", "Orange"];
let index = 0;

function showFruit() {

    // If all fruits have been displayed, reset
    if (index == fruits.length) {
        document.getElementById("result").innerHTML = "";
        index = 0;
    }

    // Display the next fruit
    document.getElementById("result").innerHTML += fruits[index] + "<br>";

    index++;
}
</script>
