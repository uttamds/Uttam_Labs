# Use Case: Remove duplicate email addresses

email_list = [
    "alice@example.com",
    "bob@example.com",
    "alice@example.com",
    "charlie@example.com",
    "bob@example.com"
]

print("Original list:")
print(email_list)

# Convert list to set to remove duplicates
unique_emails = set(email_list)

print("\nUnique email addresses:")
print(unique_emails)

# Send notification to each unique email
print("\nSending notifications:")
for email in unique_emails:
    print(f"Notification sent to {email}")
