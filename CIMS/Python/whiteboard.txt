<!DOCTYPE html>
<html>
<head>
    <title>Session Storage Demo</title>
</head>
<body>

    <h2>Session Storage Demo</h2>

    Name:
    <input type="text" id="name">

    <button onclick="saveName()">Save</button>
    <button onclick="showName()">Show</button>
    <button onclick="clearName()">Clear</button>

    <h3 id="result"></h3>

    <script>

        function saveName() {
            let name = document.getElementById("name").value;
            sessionStorage.setItem("studentName", name);
            alert("Name Saved");
        }

        function showName() {
            let name = sessionStorage.getItem("studentName");
            document.getElementById("result").innerHTML =
                "Stored Name: " + name;
        }

        function clearName() {
            sessionStorage.removeItem("studentName");
            document.getElementById("result").innerHTML = "";
            alert("Session Storage Cleared");
        }

    </script>

</body>
</html>
