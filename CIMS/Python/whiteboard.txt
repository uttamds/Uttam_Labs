<!DOCTYPE html>
<html>
<head>
    <title>Users</title>
</head>

<body>

<h2>Users</h2>

<table border="1">
    <thead>
        <tr>
            <th>ID</th>
            <th>First Name</th>
            <th>Last Name</th>
            <th>Email</th>
        </tr>
    </thead>

    <tbody id="userTable">
    </tbody>
</table>

<script>

fetch("http://127.0.0.1:5000/api/users")
    .then(response => response.json())
    .then(data => {

        let rows = "";

        data.users.forEach(user => {

            rows += `
                <tr>
                    <td>${user.id}</td>
                    <td>${user.firstName}</td>
                    <td>${user.lastName}</td>
                    <td>${user.email}</td>
                </tr>
            `;

        });

        document.getElementById("userTable").innerHTML = rows;

    });

</script>

</body>
