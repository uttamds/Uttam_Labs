# User input
full_name = "  John Doe  "

# Remove leading/trailing spaces
name = full_name.strip()

print("Original:", full_name)
print("After strip:", name)

# Convert case
print("Upper:", name.upper())
print("Lower:", name.lower())
print("Title:", name.title())

# Length
print("Length:", len(name))

# Replace text
print("Replace Doe with Smith:", name.replace("Doe", "Smith"))

# Check content
print("Starts with John?", name.startswith("John"))
print("Contains 'Doe'?", "Doe" in name)

# Split into words
words = name.split()
print("Words:", words)

# Join words
joined = "-".join(words)
print("Joined:", joined)

# String formatting
first_name = words[0]
print(f"Welcome, {first_name}!")
