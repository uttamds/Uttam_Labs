
class TravelRoute:
    def __init__(self, route_id, source, destination, fare, seats):
        self.route_id = route_id
        self.source = source
        self.destination = destination
        self.fare = fare
        self.seats = seats

    def display_route(self):
        print("Route ID      :", self.route_id)
        print("Source        :", self.source)
        print("Destination   :", self.destination)
        print("Fare          :", self.fare)
        print("Seats Available:", self.seats)
        print("-" * 35)


# Creating objects for ABC Travel Company routes
route1 = TravelRoute("R101", "Mumbai", "Pune", 800, 25)
route2 = TravelRoute("R102", "Mumbai", "Goa", 2500, 18)
route3 = TravelRoute("R103", "Pune", "Nagpur", 1800, 20)

# Displaying route details
print("ABC Travel Company - Available Routes")
print("=" * 40)

route1.display_route()
route2.display_route()
route3.display_route()

class TravelRoute:
    def __init__(self, source, destination, fare):
        self.source = source
        self.destination = destination
        self.fare = fare

    def show_route(self):
        print("Route:", self.source, "to", self.destination)
        print("Base Fare:", self.fare)


class BusRoute(TravelRoute):
    def __init__(self, source, destination, fare, bus_type):
        super().__init__(source, destination, fare)   # calling parent constructor
        self.bus_type = bus_type

    def show_route(self):
        total_fare = self.fare + 100   # extra service charge
        print("Route:", self.source, "to", self.destination)
        print("Bus Type:", self.bus_type)
        print("Final Fare:", total_fare)


# objects
r1 = TravelRoute("Mumbai", "Pune", 800)
r2 = BusRoute("Mumbai", "Goa", 2500, "AC Sleeper")

print("Parent Class Object")
r1.show_route()

print("\nChild Class Object")
r2.show_route()

class Payment:
    def pay(self, amount):
        print(f"Paying ₹{amount}")

class CreditCard(Payment):
    def pay(self, amount):
        print(f"Paid ₹{amount} using Credit Card")

class UPI(Payment):
    def pay(self, amount):
        print(f"Paid ₹{amount} using UPI")

class Cash(Payment):
    def pay(self, amount):
        print(f"Paid ₹{amount} using Cash")


# Function showing polymorphism
def make_payment(payment_method, amount):
    payment_method.pay(amount)


# Objects of different classes
cc = CreditCard()
upi = UPI()
cash = Cash()

# Same function call, different behavior
make_payment(cc, 1500)
